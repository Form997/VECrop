# VECrop Core Equations

## Water
[
\Psi_w=\Psi_p+\Psi_s+\Psi_m+\Psi_g
]
[
J_w=-K_h\nabla\Psi_w
]

## Nutrient uptake
[
J_i=V_{\max,i}^{eff}\frac{C_i}{K_{m,i}+C_i}+J_{LATS,i}-J_{efflux,i}
]

## Xylem
[
J_x=-K_{x,eff}\partial_z\Psi_x
]

## Phloem
[
Q_{ph}=K_{ph}(P_{source}-P_{sink})
]

## Carbon allocation
[
a_o=\frac{SS_o}{\sum_jSS_j}
]

## Yield
[
Y=\frac{M_{harvest}(T)}{S}
]
